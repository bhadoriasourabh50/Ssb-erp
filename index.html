    function compressImage(file, callback) {

      const reader = new FileReader();

      reader.onload = function(e) {

        const img = new Image();

        img.onload = function() {

          const canvas = document.createElement("canvas");

          const max_width = 800;

          let width = img.width;

          let height = img.height;

          if (width > max_width) {

            height = Math.round((height * max_width) / width);

            width = max_width;

          }

          canvas.width = width;

          canvas.height = height;

          const ctx = canvas.getContext("2d");

          ctx.drawImage(img, 0, 0, width, height);

          callback(canvas.toDataURL("image/jpeg", 0.65));

        };

        img.src = e.target.result;

      };

      reader.readAsDataURL(file);

    }

    function getFilteredEntries() {

      const entries = getEntries();

      const start = document.getElementById("filterStart").value;

      const end = document.getElementById("filterEnd").value;

      const partner = document.getElementById("filterPartner").value;

      const site = document.getElementById("filterSite").value;

      const party = document.getElementById("filterParty").value;

      const staff = document.getElementById("filterStaff").value;

      return entries.filter(item => {

        if(start && item.date < start) return false;

        if(end && item.date > end) return false;

        if(partner !== "ALL" && item.partnerHead !== partner) return false;

        if(site !== "ALL" && item.site !== site) return false;

        if(party !== "ALL" && item.party !== party) return false;

        if(staff !== "ALL" && !item.addedByStaff.includes(staff)) return false;

        return true;

      });

    }
